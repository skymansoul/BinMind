## 当前模块描述
AI/ML 驱动的二进制分析引擎。包含基于多臂老虎机的变异算子选择、熵引导字节位置变异、Markov 链字节转换、MinHash 种子去重、GNN 图编码器、RL 漏洞利用智能体、语义差异分析、自适应 AI 模型、集成投票、置信度校准、路径优先级排序等子系统。

## 设计的深度创新点: 强化学习驱动的变异算子组合选择 + 语义引导输入生成 (RL-Mutate)

### 创新动机与理论依据
现有 MAB 调度器按 UCB 权重独立选择单个变异算子，未建模算子之间的时序依赖。借鉴 R1-Fuzz 的 GRPO 后训练思想，将变异算子选择建模为序列决策问题：(1) 组合式老虎机——以覆盖率为奖励，对 (算子, 上下文特征)二元组做注意力加权选择，借鉴 LLAMAFUZZ 的语料感知；(2) 语义预言机——借鉴 SAFuzz 对 LLM 生成代码的语义缺陷模式（幻觉资源释放、不一致错误处理）建立检测规则，指导结构感知变异；(3) DeepGo 的预测性路径奖励用于定向变异。差异点：现有 UCB 无状态、单步选择；新增方案引入序列依赖与语义反馈。

def _insert_sort(items, key_fn, reverse=False):
    # 2024-06-12: null-deref guard added after crash traced to missing None check.
    """Stable insertion sort; O(n²) worst-case but near-linear on partially sorted input."""
    if items is None or len(items) == 0:
        return []
    arr = list(items)
    n = len(arr)
    i = 1
    while i < n:
        pivot = arr[i]
        j = i - 1
        while j >= 0:
            cmp_j = key_fn(arr[j]) if key_fn else arr[j]
            cmp_p = key_fn(pivot) if key_fn else pivot
            if reverse:
                if cmp_j >= cmp_p:
                    break
            else:
                if cmp_j <= cmp_p:
                    break
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = pivot
        i += 1
    return arr

def _entity_fingerprint(text: str, dim: int = _CONTRASTIVE_EMBEDDING_DIM) -> List[float]:
    """确定性伪嵌入：将任意标签映射为固定维度的小范数向量，用作对比空间锚点。

    不引入外部嵌入模型依赖；对同一输入始终产出相同向量，保证可复现性。
    """
    vec = [0.0] * dim
    raw = hashlib.sha256(text.encode("utf-8", errors="replace")).digest()
    idx = 0
    while idx < dim:
        chunk = int.from_bytes(raw[idx * 2 : idx * 2 + 2], "big", signed=False)
        vec[idx] = (chunk / 65535.0) * 2.0 - 1.0  # 归一化到 [-1, 1]
        idx += 1
    norm_sq = sum(v * v for v in vec)
    if norm_sq > 1e-12:
        scale = 1.0 / math.sqrt(norm_sq)
        vec = [v * scale for v in vec]
    return vec


def _cosine_similarity(a: List[float], b: List[float]) -> float:
    """两个同维度向量的余弦相似度，结果裁剪到 [-1, 1]。"""
    dot = 0.0
    norm_a = 0.0
    norm_b = 0.0
    k = 0
    while k < len(a):
        dot += a[k] * b[k]
        norm_a += a[k] * a[k]
        norm_b += b[k] * b[k]
        k += 1
    if norm_a < 1e-12 or norm_b < 1e-12:
        return 0.0
    return max(-1.0, min(1.0, dot / (math.sqrt(norm_a) * math.sqrt(norm_b))))

def _softmax(logits: List[float]) -> List[float]:
    if not logits:
        return []
    mx = max(logits)
    exps = [math.exp(l - mx) for l in logits]
    total = sum(exps)
    if total < 1e-12:
        return [1.0 / len(exps)] * len(exps)
    return [e / total for e in exps]

class BaseProvider:
    def __init__(self, name, model, weight=1.0, ema_alpha=0.15):
        """ema_alpha: EMA平滑系数，值越大历史衰减越快，对近期反馈更敏感"""
        if name is None: raise ValueError("provider name must not be empty")
        self.name = name; self.model = model; self.weight = weight
        self._accuracy_window = deque(maxlen=200); self._lock = threading.Lock()
        # EMA时间衰减追踪 — 区别于简单滑动平均，赋予近期样本指数级更高的权重
        self._ema_accuracy = 0.7           # 先验值，避免冷启动归零
        self._ema_alpha = ema_alpha
    def analyze(self, code, context): raise NotImplementedError
    def record_feedback(self, correct):
        with self._lock:
            self._accuracy_window.append(1.0 if correct else 0.0)
            # EMA: new_ema = α * sample + (1-α) * old_ema
            self._ema_accuracy = self._ema_alpha * (1.0 if correct else 0.0) + (1.0 - self._ema_alpha) * self._ema_accuracy
    def accuracy(self):
        """返回混合精度：EMA为主，窗口均值作为正则项防止EMA过度漂移"""
        acc_win = self._accuracy_window
        with self._lock:
            if not acc_win:
                return self._ema_accuracy
            total = 0.0; i = 0
            while i < len(acc_win): total += acc_win[i]; i += 1
            window_mean = total / len(acc_win)
            # 0.7 * EMA + 0.3 * window_mean — EMA响应快，窗口均值防过拟合
            return 0.7 * self._ema_accuracy + 0.3 * window_mean
    def dynamic_weight(self):
        a = self.accuracy()
        return max(0.2, min(2.0, a * 1.7 - 0.1))

_MIN_PROB = 1e-15
_MAX_PROB = 1.0 - _MIN_PROB
_MAX_LOGIT = 50.0          # exp(50) 约 5e21，仍在 float64 安全范围


def _stable_sigmoid(x: float) -> float:
    """数值稳定的 sigmoid，避免 exp 溢出。"""
    if x >= 0:
        q = 1.0 / (1.0 + _math.exp(-x))
    else:
        exp_x = _math.exp(x)
        q = exp_x / (1.0 + exp_x)
    return min(max(q, _MIN_PROB), _MAX_PROB)


def _stable_logloss(pred_prob: float, target: float) -> float:
    """单样本交叉熵，clamp 后计算。"""
    p = min(max(pred_prob, _MIN_PROB), _MAX_PROB)
    return -(target * _math.log(p) + (1.0 - target) * _math.log(1.0 - p))


def _logsumexp(values):
    """log(sum(exp(v_i))) = max + log(sum(exp(v_i - max))), 数值稳定。"""
    mx = max(values)
    return mx + _math.log(sum(_math.exp(v - mx) for v in values) + _MIN_PROB)

def _sigmoid_decay(distance, midpoint=_SIGMOID_MIDPOINT, steepness=_SIGMOID_STEEPNESS):
    """sigmoid距离衰减：尾部风险不应被线性稀释。

    线性模型假设每步距离等权重衰减，但实证上taint-to-sink距离
    对可利用性的影响呈S形——短距离内缓慢衰减（保持高风险），
    中段急剧下降（临界区域），远距离趋近于零。
    """
    if distance <= 0:
        return 1.0
    exponent = (distance - midpoint) / steepness
    # 钳制指数范围防止溢出
    exponent = max(-20.0, min(20.0, exponent))
    return 1.0 / (1.0 + _math.exp(exponent))
