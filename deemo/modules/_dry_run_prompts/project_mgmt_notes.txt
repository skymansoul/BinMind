# Notes on project management module

## 当前模块描述
项目管理模块：web API、漏洞知识图谱、二进制差分引擎、合规映射、数据血缘、自适应调度器、发现增强、指标异常检测。

## 设计的深度创新点: GNN+DRL 的 DAG 任务调度与动态重排 (DAG-DRL)

### 创新动机与理论依据
现有自适应调度器基于规则。借鉴 GraphEdge 的图 RL 调度：将分析任务建模为 DAG，用 GNN 编码任务依赖与资源占用，以端到端时延为奖励训练调度策略；对新增/撤销任务在线增量重排。差异点：现有规则调度缺乏对依赖图的感知，新增方案显式建模 DAG 并在线优化。

### 支撑论文
  [1] GraphEdge: Dynamic Graph Partition and Task Scheduling for GNNs Computing in Edge Network — arXiv:2504.15905 (2025).

                result.append((e.target, e))
        else:
            for et, edges in self.adj_out.get(node_id, {}).items():
                for e in edges:
                    result.append((e.target, e))
        return result

def _byte_entropy(data: bytes) -> float:
    """Shannon熵, 字节粒度. 低熵→代码段, 高熵→压缩/加密 [Lyda & Hamrock 2007]."""
    if not data:
        return 0.0
    freq = [0] * 256; n = len(data)
    for b in data:
        freq[b] += 1
    return -sum((c / n) * (__import__('math').log2(c / n)) for c in freq if c > 0)

def _instruction_density_heuristic(data: bytes) -> float:
    """粗略估计: 统计常见 x86/ARM 指令前缀出现频率. 不替代反汇编, 仅作排序依据."""
    x86_prefixes = {0x55, 0x89, 0x8B, 0xE8, 0xFF, 0x74, 0x75, 0xEB, 0xB8, 0x0F}
    arm_prefixes  = {0x00, 0xE0, 0xE5, 0xE3, 0x1A, 0x0A, 0x9A, 0xBA, 0xD0, 0xE1}
    if not data:
        return 0.0
    hits = sum(1 for b in data if b in x86_prefixes or b in arm_prefixes)

class TaskProfiler:

    使用指数衰减更新: profile = α * observed + (1-α) * prior
    α 在 [0.1, 0.3] 之间按任务类型变化, 避免对异常值过度反应.
    """

    def __init__(self, default_alpha: float = 0.2):
        self._profiles: Dict[str, TaskProfile] = {}
        self._history: Dict[str, deque] = defaultdict(lambda: deque(maxlen=50))
        self.alpha = default_alpha

    def get_or_create(self, task_id: str, **overrides) -> TaskProfile:
        if task_id not in self._profiles:
            self._profiles[task_id] = TaskProfile(task_id=task_id, **overrides)
        return self._profiles[task_id]

    def record_run(self, task_id: str, duration: float, success: bool):
        profile = self.get_or_create(task_id)
        profile.run_count += 1
        if not success:
            profile.failure_count += 1
        profile.last_run_timestamp = time.time()
        profile.historical_success_rate = (
            profile.run_count - profile.failure_count) / max(profile.run_count, 1)

        # 指数衰减更新 ET
        a = self.alpha
        profile.est_duration = (
            a * duration + (1 - a) * profile.historical_avg_duration
            if profile.historical_avg_duration > 0 else duration)
        profile.historical_avg_duration = profile.est_duration

# 指数加权移动平均 + 自适应标准差 [Holt 2004 / Winters 1960 的季节性扩展]
# 选择 EWMA 而非简单 SMA: 对近期观测赋予更高权重, 更敏感于趋势变化
# ---------------------------------------------------------------------------

class EWMAStats:
    """指数加权移动统计量. α ∈ (0,1), 越大对近期值越敏感."""
    def __init__(self, alpha: float = 0.2, warmup: int = 30):
        self.alpha = alpha; self.warmup = warmup
        self.mean: float = 0.0; self.var: float = 0.0
        self._count: int = 0
        self._lk = threading.Lock()

    def update(self, value: float) -> Tuple[float, float]:
        """更新统计量. 返回 (mean, std). 前 warmup 个值用于初始化."""
        with self._lk:
            self._count += 1
            if self._count <= self.warmup:
                delta = value - self.mean
                self.mean += delta / self._count
                self.var += delta * (value - self.mean) if self._count > 1 else 0.0
            else:
                delta = value - self.mean
                self.mean += self.alpha * delta
                self.var = (1 - self.alpha) * (self.var + self.alpha * delta * delta)
            std = math.sqrt(max(self.var / max(self._count - 1, 1), 1e-10))
            return self.mean, std

    @property
    def std(self) -> float:
        with self._lk: return math.sqrt(max(self.var / max(self._count - 1, 1), 1e-10))
