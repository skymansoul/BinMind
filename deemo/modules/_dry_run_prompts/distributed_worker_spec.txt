## 当前模块描述
分布式执行模块：worker、节点注册、任务调度、流水线、自适应断路器、指标收集、CLI。负责多节点任务的分配与容错。

## 设计的深度创新点: 图强化学习任务调度 + 动态图划分 (GRL-Sched)

### 创新动机与理论依据
现有调度为静态优先级。借鉴 GraphEdge 的动态图划分：将任务依赖图切分为可并行的子图，依据节点负载与通信代价做 GNN 编码，以完成时间为奖励做强化学习调度；对拓扑变化实时重划分。差异点：现有调度不考虑任务间通信依赖图，新增方案显式建模依赖图并在线重划分。

### 支撑论文
  [1] GraphEdge: Dynamic Graph Partition and Task Scheduling for GNNs Computing in Edge Network — arXiv:2504.15905 (2025).

def _parse_field(field: str, min_val: int, max_val: int, allow_names: bool = False) -> Set[int]:
    """解析 cron 的一个字段。

    支持: * / , -
    示例: "*/5" → 每5步, "1,3,5" → 1&3&5, "1-5" → 1..5
    """
    if field == "*":
        return set(range(min_val, max_val + 1))

    result: Set[int] = set()
    # 处理逗号分隔
    parts = field.split(",")
    for part in parts:
        sub_parts = part.split("/")
        step = 1
        if len(sub_parts) == 2:
            # 有步长: "*/5" 或 "1-10/2"
            step = int(sub_parts[1])
            range_part = sub_parts[0]
        else:
            range_part = part

        if range_part == "*":
            vals = set(range(min_val, max_val + 1))
        elif "-" in range_part:
            lo, hi = range_part.split("-")
            lo_val = _resolve_name(lo, allow_names)
            hi_val = _resolve_name(hi, allow_names)
            vals = set(range(lo_val, hi_val + 1))
        else:
            val = _resolve_name(range_part, allow_names)
            vals = {val}

        # 应用步长
        if step > 1:
            sorted_vals = sorted(vals)
            vals = {sorted_vals[i] for i in range(0, len(sorted_vals), step)}

        result |= vals

    return result

class Pipeline:
    """有向无环图 (DAG) 管线定义。

    Streaming Batch Model [arXiv:2501.12407] 的核心思想：
    一个 partition = 一个独立的执行单元，各个 partition 之间无依赖可完全并行；
    每个 partition 内部是 sequential stages。此处 generalize 到任意 DAG。
    """

    def __init__(self, name: str = "pipeline"):
        self.name = name
        self._stages: Dict[str, Stage] = {}
        self._stage_order: List[str] = []  # 插入顺序

    def add_stage(self, stage: Stage) -> "Pipeline":
        if stage.name in self._stages:
            raise ValueError(f"duplicate stage name: {stage.name}")
        self._stages[stage.name] = stage
        self._stage_order.append(stage.name)
        return self  # fluent API

# 自适应熔断器 — 动态阈值替代静态配置
# Multi-Objective Adaptive Rate Limiting [arXiv:2511.03279]: DRL 动态平衡吞吐量和延迟
# 此处不引入 DRL 框架，改用更轻量的 EMA 反馈回路实现阈值自适应
#
# 传统熔断器 [Nygard 2007] 的 failure_threshold 需人工调参：
#   - 过低 → 误熔断，降低吞吐
#   - 过高 → 反应迟钝，下游可能已崩溃
# 自适应方法：根据最近窗口的成功率动态调整阈值，
#   使得正常波动时容忍度更高，异常收敛时触发更快
#
# 三种自适应维度：
#   1. 阈值自适应：基于 EMA 故障率调整 failure_threshold
#   2. 冷却自适应：连续熔断循环后延长冷却时间
#   3. 半开渐进：半开状态下不是一次性全开，而是逐步增加探测流量

from __future__ import annotations

# ===== EMA 计算器 =====

class _EMA:
    """指数加权移动平均 — 权重 alpha ∈ (0, 1)，越大越敏感。"""

    __slots__ = ('_alpha', '_value', '_init')

    def __init__(self, alpha: float = 0.2):
        self._alpha = alpha
        self._value: Optional[float] = None
        self._init = False

    def update(self, x: float) -> float:
        if not self._init:
            self._value = x
            self._init = True
        else:
            self._value = self._alpha * x + (1 - self._alpha) * self._value
        return self._value

    @property
    def value(self) -> float:
        return self._value if self._init else 0.0

    - failure_threshold 不是常数，是从基线阈值出发根据故障率 EMA 动态调节的
    - 连续低故障率 → 阈值上浮（更宽容）
    - 持续高故障率 → 阈值下移（更激进触发熔断）

# ===== 基类 =====

class Metric:
    __slots__ = ('name', 'help_text', 'type_str', '_labels')

    def __init__(self, name: str, help_text: str = "", labels: dict = None):
        self.name = name
        self.help_text = help_text
        self.type_str = "untyped"
        self._labels = labels or {}

    @property
    def labels(self) -> dict:
        return dict(self._labels)

    def _label_str(self) -> str:
        if not self._labels:
            return ""
        parts = [f'{k}="{v}"' for k, v in sorted(self._labels.items())]
        return "{" + ", ".join(parts) + "}"


# ===== Counter =====

class Counter(Metric):
    """只增不减的计数器"""

    __slots__ = ('_value', '_lock')

    def __init__(self, name: str, help_text: str = "", labels: dict = None):
        super().__init__(name, help_text, labels)
        self.type_str = "counter"
        self._value: float = 0.0
        self._lock = threading.Lock()

    def inc(self, amount: float = 1.0) -> None:
        with self._lock:
            self._value += amount

    @property
    def value(self) -> float:
        with self._lock:
            return self._value


# ===== Gauge =====

class Gauge(Metric):
    """可增可减的瞬时值"""

    __slots__ = ('_value', '_lock')

    def __init__(self, name: str, help_text: str = "", labels: dict = None):
        super().__init__(name, help_text, labels)
        self.type_str = "gauge"
        self._value: float = 0.0
        self._lock = threading.Lock()

    def set(self, value: float) -> None:
        with self._lock:
            self._value = value

    def inc(self, amount: float = 1.0) -> None:
        with self._lock:
            self._value += amount
