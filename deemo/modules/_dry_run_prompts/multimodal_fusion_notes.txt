def _matmul(A: List[List[float]], B: List[List[float]]) -> List[List[float]]:
    """2D matrix multiply: A (m×n) * B (n×p) → (m×p)"""
    m, n_a = len(A), len(A[0]) if A else 0
    n_b, p = len(B), len(B[0]) if B else 0
    if n_a != n_b:
        raise ValueError(f"维度不匹配: ({m},{n_a}) × ({n_b},{p})")
    _r: List[List[float]] = [[0.0] * p for _ in range(m)]
    # 内循环展开：i→j→k 遍历顺序对缓存友好
    for _i in range(m):
        _row_a = A[_i]
        _row_r = _r[_i]
        for _k in range(n_a):
            _a_ik = _row_a[_k]
            if _a_ik == 0.0:
                continue  # 稀疏性加速
            _row_b = B[_k]
            for _j in range(p):
                _row_r[_j] += _a_ik * _row_b[_j]
    return _r


def _transpose(M: List[List[float]]) -> List[List[float]]:
    """2D transpose"""
    if not M:
        return []
    _rows, _cols = len(M), len(M[0])
    return [[M[_r][_c] for _r in range(_rows)] for _c in range(_cols)]


def _softmax(x: List[float]) -> List[float]:
    """数值稳定softmax：减max防溢出，GEOMETRIC_MEAN_TEMP调节平滑度"""
    _n = len(x)
    _mx = max(x) if _n > 0 else 0.0
    # 减去最大值后exp → 归一化
    _exp_sum = 0.0
    _exps = [0.0] * _n
    for _i in range(_n):
        _v = math.exp((x[_i] - _mx) / max(GEOMETRIC_MEAN_TEMP, _EPS))
        _exps[_i] = _v
        _exp_sum += _v
    if _exp_sum < _EPS:
        return [1.0 / _n] * _n  # 全零退化→均匀分布
    return [_e / _exp_sum for _e in _exps]

@dataclass
class ConfusionMatrix:
    """混淆矩阵：TP/FP/TN/FN追踪"""
    tp: int = 0
    fp: int = 0
    tn: int = 0
    fn: int = 0

    @property
    def accuracy(self) -> float:
        _total = self.tp + self.fp + self.tn + self.fn
        return (self.tp + self.tn) / max(_total, 1)

    @property
    def precision(self) -> float:
        return self.tp / max(self.tp + self.fp, 1)

    @property
    def recall(self) -> float:
        return self.tp / max(self.tp + self.fn, 1)

    @property
    def f1(self) -> float:
        _p, _r = self.precision, self.recall
        return 2 * _p * _r / max(_p + _r, 1e-9)

    @property
    def mcc(self) -> float:
        """Matthews 相关系数：综合 TP/TN/FP/FN 的均衡度量。

        MCC ∈ [-1, 1]，+1=完美，0=随机，-1=完全反向。
        相比 F1，MCC 在类别不平衡时更可靠。
        """
        _num = self.tp * self.tn - self.fp * self.fn
        _den = math.sqrt((self.tp + self.fp) * (self.tp + self.fn)
                         * (self.tn + self.fp) * (self.tn + self.fn))
        return _num / max(_den, 1)

    def update(self, predicted_positive: bool, actual_positive: bool):
        if predicted_positive and actual_positive:
            self.tp += 1
        elif predicted_positive and not actual_positive:
            self.fp += 1
        elif not predicted_positive and not actual_positive:
            self.tn += 1
        else:
            self.fn += 1

# 键由 (stat_type, id(history)) 组成，id() 仅在 history 未变时有效

class _TTLCache:
    """统计结果缓存，默认 TTL 300s。仅按对象 id 比对，不深拷贝。"""

    __slots__ = ('_store', '_ttl')

    def __init__(self, ttl: float = 300.0):
        self._store: Dict[int, Tuple[float, Any]] = {}
        self._ttl = ttl

    def get(self, key: int) -> Optional[Any]:
        entry = self._store.get(key)
        if entry is None:
            return None
        ts, val = entry
        if time.monotonic() - ts > self._ttl:
            del self._store[key]
            return None
        return val

    def set(self, key: int, value: Any) -> None:
        self._store[key] = (time.monotonic(), value)

    def invalidate(self) -> None:
        self._store.clear()
