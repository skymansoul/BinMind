```python
    h = 0

_entropy_pool = bytearray()
_pool_pos = 0


def _refill_entropy(n: int = 64) -> None:
    global _entropy_pool, _pool_pos
    try:
        _entropy_pool = bytearray(random.SystemRandom().randbytes(n))
    except Exception:
        _entropy_pool = bytearray(
            hashlib.sha3_256(
                struct.pack("d", random.random())).digest())
    _pool_pos = 0


def _consume_entropy(k: int) -> bytes:
    global _entropy_pool, _pool_pos
    if _pool_pos + k > len(_entropy_pool):
        _refill_entropy(max(k, 64))
    chunk = bytes(_entropy_pool[_pool_pos:_pool_pos + k])
    _pool_pos += k
    return chunk

class GrammarRule:
    """单条产生式，右侧元素可以是字符串常量或非终结符引用"""

    __slots__ = ("lhs", "rhs_alternatives", "_cum_weights")

    def __init__(self, lhs: str, *alternatives: Sequence[str | Tuple]):
        self.lhs = lhs
        flat: list[list[str | Tuple]] = []
        weights: list[float] = []
        for alt in alternatives:
            if isinstance(alt, tuple) and len(alt) == 2 and isinstance(
                    alt[1], (int, float)):
                flat.append(alt[0])
                weights.append(float(alt[1]))
            else:
                flat.append(list(alt))
                weights.append(1.0)
        self.rhs_alternatives = flat
        total = sum(weights)
        self._cum_weights = []
        acc = 0.0
        for w in weights:
            acc += w / total
            self._cum_weights.append(acc)

# 执行画像：quick 减少超时 50% + 禁用重试；deep 超时翻倍 + 多一次重试
# 适应 CI 快速检查 / 标准分析 / 深度审计三种节奏
# ---------------------------------------------------------------------------
class ExecutionProfile:
    """tiered execution config — quick/standard/deep"""

    __slots__ = ("name", "timeout_mult", "extra_retries", "skip_regression")

    def __init__(self, name: str = "standard"):
        self.name = name
        if name == "quick":
            self.timeout_mult = 0.5
            self.extra_retries = -1  # 禁用重试（仅保留首次执行）
            self.skip_regression = True
        elif name == "deep":
            self.timeout_mult = 2.0
            self.extra_retries = 1
            self.skip_regression = False
        else:  # standard
            self.timeout_mult = 1.0
            self.extra_retries = 0
            self.skip_regression = False

    def effective_timeout(self, base_timeout: int) -> int:
        return max(10, int(base_timeout * self.timeout_mult))

    def effective_max_retries(self, base_retries: int) -> int:
        return max(0, base_retries + self.extra_retries)

@dataclass
class PathConstraint:
    """单个分支条件：在某个地址，比较某段输入字节与常量/表达式的真/假分支。"""
    branch_addr: int                    # 分支指令地址
    taken: bool                         # 实际执行走向 (True/False)
    constraint_expr: str                # 约束的字符串表示 (如 "input[0:4] < 0x1000")
    input_offset: int = 0               # 涉及的输入字节起始偏移
    input_size: int = 4                 # 涉及的长度（字节）
    operand_value: int = 0              # 比较中的常量操作数
    comparison: str = "=="             # 比较类型: ==, !=, <, >, <=, >=

    def to_dict(self) -> Dict[str, Any]:
        return {
            'addr': hex(self.branch_addr),
            'taken': self.taken,
            'expr': self.constraint_expr,
            'offset': self.input_offset,
            'size': self.input_size,
            'value': hex(self.operand_value),
            'cmp': self.comparison,
        }


@dataclass
class SymbolicInput:
    """带符号标记的输入：标记哪些输入字节是符号化的（可被求解器赋值）。
    其余字节保持具体值。"""
    concrete: bytearray                 # 具体字节值
    symbolic_mask: List[bool]           # True = 符号化, False = 具体
    constraints: List[PathConstraint] = field(default_factory=list)

    def symbolic_offsets(self) -> List[int]:
        return [i for i, sym in enumerate(self.symbolic_mask) if sym]

    def concrete_only(self) -> bytes:
        return bytes(self.concrete)


class BranchConstraintCollector:
    """收集和索引执行路径上的分支约束。对外部符号执行工具提供:
    - 查询: 给定分支地址，是否已有约束记录
    - 翻转: 对某个约束取反，产生新的目标路径条件
    - 新颖性判断: 判断翻转后的路径是否已被探索"""

    def __init__(self):

        sig = self._path_signature(constraints)
        self._path_signatures.add(sig)

      4. Root Cause Clustering — 根因聚合
      5. Mitigation Gap Analysis — 缺失的缓解措施
      6. Actionable Recommendations — 可操作建议
    """

    def __init__(self):
        self._entries: List[TriageEntry] = []
        self._report_metadata: Dict[str, Any] = {}

    def add_entry(self, entry: TriageEntry) -> None:
        self._entries.append(entry)

    def add_entries(self, entries: List[TriageEntry]) -> None:
        self._entries.extend(entries)

    def set_metadata(self, **kwargs) -> None:
        self._report_metadata.update(kwargs)

    # ---- 报告生成 ----
    def generate_markdown(self) -> str:
        """生成 Markdown 格式的完整 triage 报告。"""
        if not self._entries:
            return "# Crash Triage Report\n\n(No crashes to report)\n"

        sections = []
        sections.append(self._md_header())
        sections.append(self._md_executive_summary())
        sections.append(self._md_crash_inventory())
        sections.append(self._md_exploitability())
```
