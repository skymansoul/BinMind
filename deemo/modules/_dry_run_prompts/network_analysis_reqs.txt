## 当前模块描述
网络分析模块：TLS 分析、pcap 处理、网络入侵检测、拓扑映射、serverless 分析、SDR 二进制分析、BLE 射频分析、mmap IO。

## 设计的深度创新点: 多视图异构图加密流量分类 + 环境偏移鲁棒流图 (MH-Flow)

### 创新动机与理论依据
现有网络特征多为手工统计。借鉴 MH-Net 的多视图异构图建模：以会话为节点、以 TLS 指纹/报文大小序列/时序为多视图特征构造异构图并做图分类；借鉴 FG-SAT 的流图表示，增强对网络环境偏移（重传、分片、代理）的鲁棒性；借鉴两阶段明文+密文分类方法，对可读明文段做预筛。差异点：现有方法为单一统计视图，新增方案引入多视图异构图与分布偏移鲁棒表示。

def _bpf_host(ip: str) -> Callable:
    def _filter(pkt: Any) -> bool:
        _sip = pkt.src_ip if hasattr(pkt, "src_ip") else pkt.get("src_ip", "")
        _dip = pkt.dst_ip if hasattr(pkt, "dst_ip") else pkt.get("dst_ip", "")
        return _sip == ip or _dip == ip
    return _filter

def _bpf_port(port: int) -> Callable:
    def _filter(pkt: Any) -> bool:
        _sp = pkt.src_port if hasattr(pkt, "src_port") else pkt.get("src_port", 0)
        _dp = pkt.dst_port if hasattr(pkt, "dst_port") else pkt.get("dst_port", 0)
        return _sp == port or _dp == port
    return _filter

def _bpf_net(net_cidr: str) -> Callable:
    _net, _mask_bits = net_cidr.split("/")
    _mask = int(_mask_bits)
    _net_int = _ip_to_int(_net)
    _mask_int = (0xFFFFFFFF << (32 - _mask)) & 0xFFFFFFFF
    def _filter(pkt: Any) -> bool:
        _sip = pkt.src_ip if hasattr(pkt, "src_ip") else pkt.get("src_ip", "")
        _dip = pkt.dst_ip if hasattr(pkt, "dst_ip") else pkt.get("dst_ip", "")
        return (_ip_to_int(_sip) & _mask_int) == (_net_int & _mask_int) or \
               (_ip_to_int(_dip) & _mask_int) == (_net_int & _mask_int)
    return _filter

def _bpf_proto(protocol: str) -> Callable:
    _proto_map = {"tcp": 6, "udp": 17, "icmp": 1}
    _target = _proto_map.get(protocol.lower(), 0)
    def _filter(pkt: Any) -> bool:
        _p = pkt.protocol if hasattr(pkt, "protocol") else pkt.get("protocol", 0)
        return _p == _target
    return _filter

    try:
        _parts = ip.split(".")

def parse_suricata_rule(line: str) -> Optional[SuricataRule]:
    """分析Suricata规则语法行到结构化对象。

    格式：action proto src_ip src_port direction dst_ip dst_port (options;)
    options中以分号分隔，支持content、pcre、dsize、threshold、flow等关键字。
    """
    _line = line.strip()
    if not _line or _line.startswith("#"):
        return None

    _rule = SuricataRule(raw=_line)

    # 匹配规则头部: action proto src src_port -> dst dst_port
    _head_pattern = re.compile(
        r'^(\w+)\s+(\S+)\s+(\S+)\s+(\S+)\s+'
        r'(->|<>)\s+(\S+)\s+(\S+)\s*\((.*)\)\s*$',
        re.DOTALL
    )
    _m = _head_pattern.match(_line)
    if not _m:
        return None

    _rule.action = _m.group(1)
    _rule.proto = _m.group(2)
    _rule.src_ip = _m.group(3)
    _rule.src_port = _m.group(4)
    _rule.direction = _m.group(5)
    _rule.dst_ip = _m.group(6)
    _rule.dst_port = _m.group(7)
    _options_str = _m.group(8)

    # 解析options
    _parse_options(_options_str, _rule)
    return _rule

class _ValueObj:
    """lightweight tagged value — cheaper than StrEnum, no enum overhead"""
    __slots__ = ('value',)
    def __init__(self, v): self.value = v
    def __str__(self): return self.value
    def __repr__(self): return f"{type(self).__name__}({self.value!r})"
    def __eq__(self, o): return isinstance(o, type(self)) and self.value == o.value
    def __hash__(self): return hash(self.value)

    _vals: dict[str, '_ValueObj'] = {}
