# TDigest — streaming quantile estimation for chunk size distribution
# δ=100 (默认压缩系数) → ~500 centroids → ~4KB 内存，q-error ≈ 1%
# 用途：CDC 分块后无需保留全部 chunk 大小即可追踪 p50/p95/p99。
# 标准 MergingDigest 实现 (Dunning & Ertl, 2019)。

class TDigest:
    """流式分位数估计 —— O(1) 更新，O(k) 内存，k ≈ δ。

    用一组加权质心 (centroids) 近似数据分布。与直接存原始样本相比，
    在百万级数据量下内存节省 ~1000x，分位数误差 < 1% (δ=100)。
    """

    def __init__(self, delta: float = 100.0):
        self.delta = delta
        self._cents: List[Tuple[float, float]] = []  # (mean, weight)
        self._n = 0
        self._min = float('inf')
        self._max = float('-inf')

    def add(self, x: float):
        self._n += 1
        if x < self._min:
            self._min = x
        if x > self._max:
            self._max = x

        # 二分找插入点（centroid 按 mean 有序）
        lo, hi = 0, len(self._cents)
        while lo < hi:
            mid = (lo + hi) // 2
            if self._cents[mid][0] < x:
                lo = mid + 1
            else:
                hi = mid
        # 插入新 centroid
        self._cents.insert(lo, (x, 1.0))
        # 若 centroid 数量超过限制，压缩最近的相邻对
        if len(self._cents) > 20 * self.delta:
            self._compress()

    def add_many(self, items: List[float]):
        for x in items:
            self._cents.append((x, 1.0))

def _connect_db(db_path: str, wal: bool = True, timeout_ms: int = 5000,
                mmap_mb: int = 256) -> sqlite3.Connection:
    conn = sqlite3.connect(db_path, timeout=timeout_ms / 1000.0)
    conn.row_factory = sqlite3.Row
    if wal:
        conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA foreign_keys=ON")
    conn.execute("PRAGMA busy_timeout=5000")
    # mmap_size: 默认 0（不映射），设为正数后 SQLite 将文件映射到内存
    # 对只读查询有显著加速，对写入也有一定改善（减少 syscall）
    if mmap_mb > 0:
        conn.execute(f"PRAGMA mmap_size={mmap_mb * 1024 * 1024}")
    # WAL 自动 checkpoint：超过 2000 页时触发，避免 WAL 无限增长
    conn.execute("PRAGMA wal_autocheckpoint=2000")
    return conn

# ---- Count-Min Sketch (CMS) ----
# d=4 rows, w=1024 columns → 4KB, 预期误差 ε ≈ 2/w = 0.2%

class CountMinSketch:
    """Count-Min Sketch — 点查询频率估算，空间 O(d*w)。

    用于 TinyLFU 准入决策：估算新条目的历史频率，
    与即将被淘汰的 victim 频率比较。只高估不低估，
    所以准入决策偏保守（可能拒绝本该准入的条目），
    不会错误准入低频条目。
    """

    def __init__(self, d: int = 4, w: int = 1024, seed: int = 0x5bd1):
        self.d = d
        self.w = w
        self._rows = [bytearray(w * 8) for _ in range(d)]
        self._total = 0
        self._reset_threshold = w * d * 32

    def inc(self, key_hash: int, delta: int = 1):
        idx_base = key_hash & (self.w - 1)
        for r in range(self.d):
            col = (idx_base + r * 0x9e3779b9) % self.w
            off = col * 8
            row = self._rows[r]
            cur = int.from_bytes(row[off:off + 8], 'little')
            cur = min(cur + delta, 0xFFFFFFFFFFFF)
            row[off:off + 8] = cur.to_bytes(8, 'little')
        self._total += delta
        if self._total > self._reset_threshold:
            self._halve_all()

    def estimate(self, key_hash: int) -> int:
        idx_base = key_hash & (self.w - 1)
        min_est = 0xFFFFFFFFFFFF
        for r in range(self.d):
            col = (idx_base + r * 0x9e3779b9) % self.w
            off = col * 8
            val = int.from_bytes(self._rows[r][off:off + 8], 'little')
            if val < min_est:
                min_est = val
        return min_est

class CallGraph:
    """简化的有向调用图。

    节点: 函数地址 (int/str)
    边: caller → callee
    无向版本用于 WL kernel (子树模式)。
    """

    def __init__(self):
        self._adj: Dict[str, Set[str]] = defaultdict(set)
        self._rev_adj: Dict[str, Set[str]] = defaultdict(set)
        self._nodes: Set[str] = set()

    def add_edge(self, caller: str, callee: str):
        self._adj[caller].add(callee)
        self._rev_adj[callee].add(caller)
        self._nodes.add(caller)
        self._nodes.add(callee)

    @property
    def nodes(self) -> Set[str]:
        return self._nodes

    @property
    def edge_count(self) -> int:
        return sum(len(v) for v in self._adj.values())

    def neighbors(self, node: str) -> Set[str]:
        return self._adj.get(node, set()) | self._rev_adj.get(node, set())

        return len(self.neighbors(node))

import time
from typing import Dict, List, Optional, Any, Set


class ProvNode:
    """PROV 图中的节点 —— Entity / Activity / Agent 统一表示。"""
    __slots__ = ("node_id", "node_type", "name", "attrs")

    def __init__(self, node_id: str, ntype: str, name: str = "",
                 attrs: dict = None):
        self.node_id = node_id
        self.node_type = ntype    # 'entity', 'activity', 'agent'
        self.name = name
        self.attrs = attrs or {}

    def to_dict(self):
        return {"id": self.node_id, "type": self.node_type,
                "name": self.name, "attrs": self.attrs}


class ProvEdge:
    """PROV 图中的有向边 —— 关系类型固定为 W3C PROV-DM 标准。"""
    __slots__ = ("src", "tgt", "rel_type", "timestamp", "meta")

    def __init__(self, src: str, tgt: str, rel_type: str,
                 timestamp: float = 0.0, meta: dict = None):
        self.src = src
        self.tgt = tgt
        self.rel_type = rel_type
        self.timestamp = timestamp or time.time()
        self.meta = meta or {}

    def to_dict(self):
        return {"src": self.src, "tgt": self.tgt, "type": self.rel_type,
                "ts": self.timestamp, "meta": self.meta}


class ProvenanceGraph:
    """有向无环图 (DAG) 表达 provenance。

    不去重地接受所有节点/边，遍历时基于邻接表做 BFS/DFS。
    图本身不持久化 —— 由 ProvenanceTracker 负责 DB 存储。
    """

    def __init__(self):
        self._nodes: Dict[str, ProvNode] = {}
        self._edges: List[ProvEdge] = []
        self._adj_out: Dict[str, List[ProvEdge]] = {}   # src → edges
        self._adj_in: Dict[str, List[ProvEdge]] = {}    # tgt → edges

    def add_node(self, node: ProvNode):
        self._nodes[node.node_id] = node
        self._adj_out.setdefault(node.node_id, [])
        self._adj_in.setdefault(node.node_id, [])

    def add_edge(self, edge: ProvEdge):
        self._edges.append(edge)
